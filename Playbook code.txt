            // Real-Time Cloud Storage Sync (RULE 1 & RULE 2 Compliant)
            setupRealtimeSync() {
                if (!this.user) return;
                
                // RULE 1 Path: /artifacts/{appId}/users/{userId}/plays
                const playsCollectionRef = collection(db, 'artifacts', appId, 'users', this.user.uid, 'plays');


                // Real-time snapshot listener
                onSnapshot(playsCollectionRef, (snapshot) => {
                    const loadedPlays = [];
                    snapshot.forEach(docSnap => {
                        loadedPlays.push({ id: docSnap.id, ...docSnap.data() });
                    });


                    if (loadedPlays.length === 0) {
                        // Populate default initial plays into cloud
                        DEFAULT_PLAYS.forEach(p => {
                            const dRef = doc(db, 'artifacts', appId, 'users', this.user.uid, 'plays', p.id);
                            setDoc(dRef, p);
                        });
                        this.plays = [...DEFAULT_PLAYS];
                    } else {
                        this.plays = loadedPlays;
                    }


                    this.populatePlaySelect();
                    if (this.plays.length > 0 && !this.currentPlay) {
                        this.loadPlay(this.plays[0].id);
                    }
                }, (error) => {
                    console.error("Cloud Sync Error:", error);
                });
            }


            async syncPlayToCloud(playObj) {
                if (!this.user || !playObj) return;
                try {
                    const playRef = doc(db, 'artifacts', appId, 'users', this.user.uid, 'plays', playObj.id);
                    await setDoc(playRef, playObj);
                } catch(err) {
                    console.error("Failed to sync play:", err);
                }
            }


            async deletePlayFromCloud(playId) {
                if (!this.user || !playId) return;
                try {
                    const playRef = doc(db, 'artifacts', appId, 'users', this.user.uid, 'plays', playId);
                    await deleteDoc(playRef);
                } catch(err) {
                    console.error("Failed to delete play:", err);
                }
            }

                });
            }

                    document.getElementById('btnModeMove').className = 'mode-btn bg-slate-800 text-slate-300 hover:bg-slate-700 p-2 rounded text-xs font-semibold flex items-center justify-center gap-1.5';
                    this.modeDesc.textContent = 'Select a drawing tool below to sketch routes, blocks, or motion paths onto the field.';
                });


                document.getElementById('btnPlay').addEventListener('click', () => this.isPlaying = true);
                document.getElementById('btnPause').addEventListener('click', () => this.isPlaying = false);
                document.getElementById('btnReset').addEventListener('click', () => this.resetAnimation());


                this.scrubber.addEventListener('input', (e) => {
                    this.progress = parseFloat(e.target.value) / 100;
                    this.timeDisplay.textContent = (this.progress * 4.0).toFixed(1) + 's';
                    this.render();
                });


                const toolBtns = [
                    { id: 'btnToolDraw', tool: 'draw' },
                    { id: 'btnToolArrow', tool: 'arrow' },
                    { id: 'btnToolBlock', tool: 'block' },
                    { id: 'btnToolMotion', tool: 'motion' },
                    { id: 'btnToolStunt', tool: 'stunt' }
                ];
                toolBtns.forEach(tb => {
                    document.getElementById(tb.id).addEventListener('click', () => {
                        this.interactionMode = 'draw';
                        this.activeTool = tb.tool;
                        toolBtns.forEach(b => document.getElementById(b.id).classList.remove('bg-blue-600', 'text-white'));
                        document.getElementById(tb.id).classList.add('bg-blue-600', 'text-white');
                    });
                });


                document.getElementById('btnToolClear').addEventListener('click', () => {
                    this.drawnLines = [];
                    this.render();
                });


                document.querySelectorAll('.color-btn').forEach(btn => {
                    btn.addEventListener('click', (e) => {
                        document.querySelectorAll('.color-btn').forEach(b => b.classList.remove('ring-2', 'ring-white'));
                        e.target.classList.add('ring-2', 'ring-white');
                        this.penColor = e.target.dataset.color;
                    });
                });


                document.getElementById('btnOpenNewPlayModal').addEventListener('click', () => {
                    this.newPlayModal.classList.remove('hidden');
                });
                document.getElementById('btnCloseModal').addEventListener('click', () => {
                    this.newPlayModal.classList.add('hidden');
                });
                document.getElementById('btnCancelModal').addEventListener('click', () => {
                    this.newPlayModal.classList.add('hidden');
                });


                document.getElementById('btnCreatePlaySubmit').addEventListener('click', async () => {
                    const name = this.modalPlayName.value.trim() || 'Custom Play';
                    const category = this.modalCategory.value;
                    const offKey = this.modalOffensePreset.value;
                    const defKey = this.modalDefensePreset.value;


                    const newPlayObj = {
                        id: 'play_' + Date.now(),
                        name: name,
                        category: category,
                        desc: `Custom created ${category} play setup.`,
                        qbKeys: ['Scan defense pre-snap', 'Execute progression timing'],
                        offense: JSON.parse(JSON.stringify(OFFENSE_PRESETS[offKey] || OFFENSE_PRESETS.blank))
                    };


                    await this.syncPlayToCloud(newPlayObj);
                    this.currentPlay = newPlayObj;
                    this.loadPlay(newPlayObj.id);


                    if (DEFENSIVE_FRONTS[defKey]) {
                        this.selectedFront = defKey;
                        this.frontSelect.value = defKey;
                        this.currentDefense = JSON.parse(JSON.stringify(DEFENSIVE_FRONTS[defKey]));
                    }


                    this.drawnLines = [];
                    this.newPlayModal.classList.add('hidden');
                });


                document.getElementById('btnSavePlay').addEventListener('click', async () => {
                    if (this.currentPlay) {
                        await this.syncPlayToCloud(this.currentPlay);
                        alert('Play synced to cloud across all devices!');
                    }
                });


                document.getElementById('btnDeletePlay').addEventListener('click', async () => {
                    if (this.plays.length > 1 && this.currentPlay) {
                        await this.deletePlayFromCloud(this.currentPlay.id);
                        this.currentPlay = null;
                    } else {
                        alert('Cannot delete the last remaining play.');
                    }
                });


                this.canvas.addEventListener('mousedown', (e) => this.handlePointerDown(e));
                this.canvas.addEventListener('mousemove', (e) => this.handlePointerMove(e));
                this.canvas.addEventListener('mouseup', () => this.handlePointerUp());


                this.canvas.addEventListener('touchstart', (e) => {
                    e.preventDefault();
                    this.handlePointerDown(e.touches[0]);
                });
                this.canvas.addEventListener('touchmove', (e) => {
                    e.preventDefault();
                    this.handlePointerMove(e.touches[0]);
                });
                this.canvas.addEventListener('touchend', () => this.handlePointerUp());
            }


            handlePointerDown(e) {
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                const w = this.canvas.width;
                const h = this.canvas.height;


                if (this.interactionMode === 'move') {
                    if (this.currentPlay && this.currentPlay.offense) {
                        for (let p of this.currentPlay.offense) {
                            const px = p.x * w;
                            const py = p.y * h;
                            const dist = Math.hypot(x - px, y - py);
                            if (dist <= 18) {
                                this.draggedPlayer = { type: 'offense', player: p };
                                return;
                            }
                        }
                    }


                    if (this.currentDefense) {
                        for (let d of this.currentDefense) {
                            const dx = d.x * w;
                            const dy = d.y * h;
                            const dist = Math.hypot(x - dx, y - dy);
                            if (dist <= 18) {
                                this.draggedPlayer = { type: 'defense', player: d };
                                return;
                            }
                        }
                    }
                } else if (this.interactionMode === 'draw') {
                    this.isDrawing = true;
                    this.currentLine = {
                        color: this.penColor,
                        type: this.activeTool,
                        points: [{x, y}]
                    };
                }
            }


            handlePointerMove(e) {
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                const w = this.canvas.width;
                const h = this.canvas.height;


                if (this.draggedPlayer) {
                    this.draggedPlayer.player.x = Math.max(0.02, Math.min(0.98, x / w));
                    this.draggedPlayer.player.y = Math.max(0.02, Math.min(0.98, y / h));
                    this.render();
                } else if (this.isDrawing && this.currentLine) {
                    if (this.activeTool === 'arrow' || this.activeTool === 'block' || this.activeTool === 'motion') {
                        this.currentLine.points = [this.currentLine.points[0], {x, y}];
                    } else {
                        this.currentLine.points.push({x, y});
                    }
                    this.render();
                }
            }


            handlePointerUp() {
                if (this.draggedPlayer) {
                    this.draggedPlayer = null;
                }
                if (this.isDrawing && this.currentLine) {
                    this.drawnLines.push(this.currentLine);
                    this.currentLine = null;
                }
                this.isDrawing = false;
                this.render();
            }


            resetAnimation() {
                this.isPlaying = false;
                this.progress = 0;
                this.scrubber.value = 0;
                this.timeDisplay.textContent = '0.0s';
                this.render();
            }


            startAnimationLoop() {
                let lastTime = performance.now();
                const animate = (now) => {
                    const dt = (now - lastTime) / 1000;
                    lastTime = now;


                    if (this.isPlaying) {
                        this.progress += (dt * 0.25) * this.speed;
                        if (this.progress >= 1.0) {
                            this.progress = 1.0;
                            this.isPlaying = false;
                        }
                        this.scrubber.value = this.progress * 100;
                        this.timeDisplay.textContent = (this.progress * 4.0).toFixed(1) + 's';
                        this.render();
                    }
                    requestAnimationFrame(animate);
                };
                requestAnimationFrame(animate);
            }


            render() {
                if (!this.canvas || !this.ctx) return;
                const w = this.canvas.width;
                const h = this.canvas.height;


                this.ctx.fillStyle = '#143d23';
                this.ctx.fillRect(0, 0, w, h);


                this.ctx.strokeStyle = '#1e5330';
                this.ctx.lineWidth = 2;
                const numYardLines = 8;
                for (let i = 1; i < numYardLines; i++) {
                    const y = (h / numYardLines) * i;
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y);
                    this.ctx.lineTo(w, y);
                    this.ctx.stroke();
                }


                this.renderHashMarks(w, h, numYardLines);


                const losY = h * 0.58;
                const fdY = h * 0.35;


                this.ctx.strokeStyle = '#f59e0b';
                this.ctx.lineWidth = 3;
                this.ctx.setLineDash([6, 6]);
                this.ctx.beginPath();
                this.ctx.moveTo(0, fdY);
                this.ctx.lineTo(w, fdY);
                this.ctx.stroke();
                this.ctx.setLineDash([]);


                this.ctx.strokeStyle = '#3b82f6';
                this.ctx.lineWidth = 3;
                this.ctx.beginPath();
                this.ctx.moveTo(0, losY);
                this.ctx.lineTo(w, losY);
                this.ctx.stroke();


                if (this.chkRoutes.checked && this.currentPlay && this.currentPlay.offense) {
                    this.currentPlay.offense.forEach(p => {
                        if (p.route && p.route.length > 1) {
                            this.ctx.strokeStyle = '#3b82f6';
                            this.ctx.lineWidth = 3;
                            this.ctx.beginPath();
                            p.route.forEach((pt, idx) => {
                                if (idx === 0) this.ctx.moveTo(pt.x * w, pt.y * h);
                                else this.ctx.lineTo(pt.x * w, pt.y * h);
                            });
                            this.ctx.stroke();
                            const lastPt = p.route[p.route.length - 1];
                            const prevPt = p.route[p.route.length - 2];
                            this.drawArrowhead(prevPt.x * w, prevPt.y * h, lastPt.x * w, lastPt.y * h, '#3b82f6');
                        }
                    });
                }


                if (this.currentDefense) {
                    this.currentDefense.forEach(d => {
                        const pos = { x: d.x * w, y: d.y * h };
                        this.ctx.fillStyle = '#dc2626';
                        this.ctx.strokeStyle = '#ffffff';
                        this.ctx.lineWidth = 2;
                        this.ctx.beginPath();
                        this.ctx.rect(pos.x - 12, pos.y - 12, 24, 24);
                        this.ctx.fill();
                        this.ctx.stroke();


                        this.ctx.fillStyle = '#ffffff';
                        this.ctx.font = 'bold 10px system-ui';
                        this.ctx.textAlign = 'center';
                        this.ctx.textBaseline = 'middle';
                        this.ctx.fillText(d.label, pos.x, pos.y);
                    });
                }


                if (this.currentPlay && this.currentPlay.offense) {
                    this.currentPlay.offense.forEach(p => {
                        let pos = { x: p.x * w, y: p.y * h };


                        if (p.motion && this.progress <= 0.3) {
                            pos = this.getPointAlongPath(p.motion, this.progress / 0.3, w, h);
                        } else if (p.route && this.progress > 0.3) {
                            pos = this.getPointAlongPath(p.route, (this.progress - 0.3) / 0.7, w, h);
                        }


                        this.ctx.fillStyle = '#2563eb';
                        this.ctx.strokeStyle = '#ffffff';
                        this.ctx.lineWidth = 2;
                        this.ctx.beginPath();
                        this.ctx.arc(pos.x, pos.y, 14, 0, Math.PI * 2);
                        this.ctx.fill();
                        this.ctx.stroke();


                        this.ctx.fillStyle = '#ffffff';
                        this.ctx.font = 'bold 11px system-ui';
                        this.ctx.textAlign = 'center';
                        this.ctx.textBaseline = 'middle';
                        this.ctx.fillText(p.label, pos.x, pos.y);
                    });
                }


                this.renderTelestrator();
            }


            renderHashMarks(w, h, numLines) {
                let leftHashX, rightHashX;
                if (this.selectedLevel === 'highschool') {
                    leftHashX = w * 0.33;
                    rightHashX = w * 0.67;
                } else if (this.selectedLevel === 'college') {
                    leftHashX = w * 0.40;
                    rightHashX = w * 0.60;
                } else {
                    leftHashX = w * 0.46;
                    rightHashX = w * 0.54;
                }


                this.ctx.strokeStyle = '#ffffff';
                this.ctx.lineWidth = 1.5;


                for (let i = 1; i < numLines * 4; i++) {
                    const y = (h / (numLines * 4)) * i;
                    this.ctx.beginPath();
                    this.ctx.moveTo(leftHashX - 5, y);
                    this.ctx.lineTo(leftHashX + 5, y);
                    this.ctx.stroke();


                    this.ctx.beginPath();
                    this.ctx.moveTo(rightHashX - 5, y);
                    this.ctx.lineTo(rightHashX + 5, y);
                    this.ctx.stroke();
                }
            }


            renderTelestrator() {
                const lines = [...this.drawnLines];
                if (this.currentLine) lines.push(this.currentLine);


                lines.forEach(line => {
                    if (line.points.length < 2) return;
                    this.ctx.strokeStyle = line.color;
                    this.ctx.lineWidth = 3;


                    if (line.type === 'motion') {
                        this.ctx.setLineDash([6, 6]);
                    } else if (line.type === 'stunt') {
                        this.ctx.setLineDash([4, 4]);
                        this.ctx.lineWidth = 4;
                    } else {
                        this.ctx.setLineDash([]);
                    }


                    this.ctx.beginPath();
                    line.points.forEach((pt, idx) => {
                        if (idx === 0) this.ctx.moveTo(pt.x, pt.y);
                        else this.ctx.lineTo(pt.x, pt.y);
                    });
                    this.ctx.stroke();
                    this.ctx.setLineDash([]);


                    const lastPt = line.points[line.points.length - 1];
                    const prevPt = line.points[line.points.length - 2];


                    if (line.type === 'arrow' || line.type === 'motion' || line.type === 'stunt') {
                        this.drawArrowhead(prevPt.x, prevPt.y, lastPt.x, lastPt.y, line.color);
                    } else if (line.type === 'block') {
                        this.drawBlockCap(prevPt.x, prevPt.y, lastPt.x, lastPt.y, line.color);
                    }
                });
            }


            getPointAlongPath(pts, ratio, w, h) {
                if (!pts || pts.length === 0) return { x: 0, y: 0 };
                if (pts.length === 1 || ratio <= 0) return { x: pts[0].x * w, y: pts[0].y * h };
                if (ratio >= 1) return { x: pts[pts.length - 1].x * w, y: pts[pts.length - 1].y * h };


                const totalSegments = pts.length - 1;
                const scaledRatio = ratio * totalSegments;
                const segIdx = Math.floor(scaledRatio);
                const segRatio = scaledRatio - segIdx;


                const p1 = pts[segIdx];
                const p2 = pts[segIdx + 1];


                return {
                    x: (p1.x + (p2.x - p1.x) * segRatio) * w,
                    y: (p1.y + (p2.y - p1.y) * segRatio) * h
                };
            }


            drawArrowhead(fromX, fromY, toX, toY, color) {
                const headlen = 12;
                const angle = Math.atan2(toY - fromY, toX - fromX);
                this.ctx.fillStyle = color;
                this.ctx.beginPath();
                this.ctx.moveTo(toX, toY);
                this.ctx.lineTo(toX - headlen * Math.cos(angle - Math.PI / 6), toY - headlen * Math.sin(angle - Math.PI / 6));
                this.ctx.lineTo(toX - headlen * Math.cos(angle + Math.PI / 6), toY - headlen * Math.sin(angle + Math.PI / 6));
                this.ctx.closePath();
                this.ctx.fill();
            }


            drawBlockCap(fromX, fromY, toX, toY, color) {
                const capLen = 12;
                const angle = Math.atan2(toY - fromY, toX - fromX);
                const perp = angle + Math.PI / 2;


                const x1 = toX + capLen * Math.cos(perp);
                const y1 = toY + capLen * Math.sin(perp);
                const x2 = toX - capLen * Math.cos(perp);
                const y2 = toY - capLen * Math.sin(perp);


                this.ctx.strokeStyle = color;
                this.ctx.lineWidth = 4;
                this.ctx.beginPath();
                this.ctx.moveTo(x1, y1);
                this.ctx.lineTo(x2, y2);
                this.ctx.stroke();
            }
        }
